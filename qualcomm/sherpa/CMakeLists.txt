function(kig_replace variable before after)
  string(FIND "${${variable}}" "${before}" found)
  if(found LESS 0)
    message(FATAL_ERROR "Pinned sherpa patch anchor missing: ${before}")
  endif()
  string(REPLACE "${before}" "${after}" value "${${variable}}")
  set(${variable} "${value}" PARENT_SCOPE)
endfunction()

set(patched_sources)

  if(source STREQUAL "vocos-vocoder.cc" OR source STREQUAL "offline-tts-zipvoice-model.cc" OR source STREQUAL "offline-tts-impl.cc")
    list(APPEND patched_sources "${CMAKE_CURRENT_BINARY_DIR}/patched/${source}")

endforeach()
